\section{Main Results}
\label{sec:main}

The four contributions of Sections~\ref{sec:adaptive}--\ref{sec:shadow} compose into a single resource profile.  We summarize the combined sample-complexity bound and its classical lower-bound preservation.

\begin{corollary}[Combined bound]\label{cor:combined}
Combining Theorems~\ref{thm:adaptive}, \ref{thm:hier}, and \ref{thm:var}, for a $K_F$-Fourier-sparse function with $K$-sparse Boolean support:
\[
  M_{\mathrm{total}} = O\!\left(\frac{K_F\cdot Q^{2-1/k}}{\eps^2}\right),
\]
an improvement of $(N/K_F)\cdot Q^{1/k}$ over the \Zhao\ baseline.
\end{corollary}

\noindent\textbf{Concrete example.}  Set $N=2^{20}$ ($\approx 10^6$), $K_F=64$, $Q=100$, $k=3$:
\begin{align*}
  \text{\Zhao:} \quad & M = N\cdot Q^2 = 2^{20}\cdot 10^4 = 1.05\times 10^{10},\\
  \text{Combined:} \quad & M = K_F\cdot Q^{2-1/3} = 64\cdot 10^{5.33} = 1.37\times 10^5.\\
  \text{Improvement:} \quad & \frac{N}{K_F}\cdot Q^{1/3} = 16{,}384\times 4.64 = 76{,}019\times.
\end{align*}

\begin{proposition}[Classical lower bound preservation]\label{prop:lb}
The adaptive and hierarchical reductions lower the \emph{quantum} oracle-construction cost without weakening the classical communication lower bounds of \Zhao.  Variational warmstart can improve quantum approximation accuracy on Fourier-sparse inputs without changing the output model.  The interferometric shadow extension estimates the same observables with unchanged classical hardness.
\end{proposition}

\begin{proof}[Proof sketch]
Adaptive and hierarchical methods replace uniform sampling with importance-weighted or residual-targeted sampling; they do not grant additional classical query power.  Warmstart preconditions the signal in Fourier space before sketching; classical lower bounds on streaming or sparse access remain.  Shadow estimators are unbiased for $\langle v|\rho|v\rangle$ with variance $O(s/T)$; the classical communication model is unchanged.
\end{proof}
